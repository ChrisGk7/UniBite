DROP DATABASE if exists unibite_db;
CREATE DATABASE unibite_db;
USE unibite_db;


CREATE TABLE user(
  username varchar(63) NOT NULL UNIQUE,
  email varchar(63) NOT NULL UNIQUE,
  pass varchar(255) DEFAULT NULL,
  name varchar(63) DEFAULT NULL,
  reg_date datetime NOT NULL DEFAULT current_timestamp(),

 
  PRIMARY KEY (username)
);

CREATE TABLE student (
  username varchar(63) NOT NULL UNIQUE,
  email varchar(63) NOT NULL UNIQUE,
  credits int(5) NOT NULL DEFAULT 5,
  street text NOT NULL,
  number INT(5) NOT NULL,
  city text NOT NULL,
  postcode INT(11) NOT NULL,
  mobile varchar(14) NOT NULL,

  FOREIGN KEY (username) REFERENCES user(username)
  ON UPDATE CASCADE ON DELETE CASCADE ,

  FOREIGN KEY (email) REFERENCES user(email)
  ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE cook(
  username varchar(63) NOT NULL UNIQUE,
  email varchar(63) NOT NULL UNIQUE,
  street text NOT NULL,
  number int(5) NOT NULL,
  city text NOT NULL,
  postcode INT(11) NOT NULL,
  mobile varchar(14) NOT NULL,

  FOREIGN KEY (email) REFERENCES user(email)
  ON UPDATE CASCADE ON DELETE CASCADE,
        
  FOREIGN KEY (username) REFERENCES user(username)
  ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE admin (
  email varchar(63) NOT NULL UNIQUE,
  username varchar(63) NOT NULL UNIQUE,

  FOREIGN KEY (username) REFERENCES user(username)
  ON UPDATE CASCADE ON DELETE CASCADE,

  FOREIGN KEY (email) REFERENCES user(email)
  ON UPDATE CASCADE ON DELETE CASCADE
);


CREATE TABLE dish(
  id int(30) NOT NULL AUTO_INCREMENT UNIQUE,
  cook varchar(63) NOT NULL,
  title varchar(255) NOT NULL,
  description varchar(255) NOT NULL,
  allergens text NOT NULL,
  photos_url varchar (255) NOT NULL DEFAULT 'C:\xampp\htdocs\unibite', 
  portions int(2) NOT NULL,
  credit_cost int (2) NOT NULL DEFAULT 1,

  reg_date datetime NOT NULL DEFAULT current_timestamp(),
  
  PRIMARY KEY (id),

  FOREIGN KEY (cook) REFERENCES cook(username)
  ON UPDATE CASCADE ON DELETE CASCADE
);


CREATE TABLE student_cook_dish_relation (
  id int(30) NOT NULL AUTO_INCREMENT UNIQUE,
  stu_usernanme varchar(63) NOT NULL,
  dish_id int(30) NOT NULL,
  cook_username varchar(63),
  status enum('pending_assignment','active','finished','canceled') DEFAULT 'pending_assignment',
  json JSON DEFAULT NULL,
  rating enum('1','2','3','4','5','pending') DEFAULT 'pending' NOT NULL ,

  PRIMARY KEY (id),

  FOREIGN KEY (stu_usernanme) REFERENCES student(username)
  ON UPDATE CASCADE ON DELETE CASCADE,

  FOREIGN KEY (cook_username) REFERENCES cook(username)
  ON UPDATE CASCADE ON DELETE CASCADE,

  FOREIGN KEY (dish_id) REFERENCES dish(id)
  ON UPDATE CASCADE ON DELETE CASCADE
);

CREATE TABLE request(
  stu_username VARCHAR(63) NOT NULL,
  cook_username VARCHAR(63) NOT NULL,
  dish_id INT(30) NOT NULL,
  portions INT(2) NOT NULL,
  status ENUM ('pending', 'declined', 'accepted') DEFAULT 'pending',
  request_datetime DATETIME NOT NULL DEFAULT current_timestamp(),
  reply_datetime DATETIME DEFAULT NULL,
  

  
  FOREIGN KEY (stu_username) REFERENCES student(username)
  ON UPDATE CASCADE ON DELETE CASCADE,

  FOREIGN KEY (cook_username) REFERENCES cook(username)
  ON UPDATE CASCADE ON DELETE CASCADE,

  FOREIGN KEY (dish_id) REFERENCES dish(id)
  ON UPDATE CASCADE ON DELETE CASCADE 
);
